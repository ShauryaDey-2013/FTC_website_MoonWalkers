MoonWalkersAutoDrive.java
FTC Java OpMode 
package org.firstinspires.ftc.team38056.opmodes;

import com.qualcomm.robotcore.eventloop.opmode.Autonomous;
import com.qualcomm.robotcore.eventloop.opmode.LinearOpMode;
import org.moonwalkers.kinematics.SwerveDriveSubsystem;
import org.moonwalkers.pathfinding.Pose2d;

@Autonomous(name = "MoonWalkers: Auto Lunar Scribe", group = "Autonomous")
public class MoonWalkersAutoDrive extends LinearOpMode {

    private SwerveDriveSubsystem drive;

    @Override
    public void runOpMode() throws InterruptedException {
        // Initialize custom 4-wheel independent swerve drive
        drive = new SwerveDriveSubsystem(hardwareMap);
        drive.setPoseEstimate(new Pose2d(0, 0, Math.toRadians(0)));

        telemetry.addData("Status", "MoonWalkers Systems Calibrated [38056]");
        telemetry.update();

        waitForStart();

        if (isStopRequested()) return;

        // Execute trajectory to target scoring zone with AprilTag vision alignment
        drive.followPathSpline(
            new Pose2d(48.0, 24.0, Math.toRadians(90.0))
        );
    }
}
